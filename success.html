/* ====================================================== MODULE — RECUP SESSION ID ====================================================== */ function getSessionId(){ const params = new URLSearchParams(window.location.search); return params.get("session_id"); } /* ====================================================== MODULE — ACTIVATION ====================================================== */ async function activer(){ const sessionId = getSessionId(); if(!sessionId){ document.getElementById("status").innerText = "❌ Session Stripe manquante"; return; } try{ // 1. récupérer email via serveur const resSession = await fetch(API + "/session/" + sessionId); const sessionData = await resSession.json(); const email = sessionData.email; if(!email){ throw new Error("Email introuvable"); } // 2. récupérer licence const res = await fetch(API + "/licence/" + email); if(!res.ok){ throw new Error("Licence introuvable"); } const data = await res.json(); localStorage.setItem("licence_key", data.cle); document.getElementById("status").innerText = "Licence activée ✔\n" + data.cle; }catch(e){ document.getElementById("status").innerText = "❌ " + e.message; } }
